# 1. Check what ALLOWED_TASK_ARNS is actually set to in the Lambda
aws lambda get-function-configuration \
  --function-name c9migration-sbx-datasync-archiver \
  --region us-west-2 \
  --query 'Environment.Variables.ALLOWED_TASK_ARNS'

# 2. Check the actual task ARN from DataSync
aws datasync list-tasks \
  --region us-west-2 \
  --query 'Tasks[].TaskArn'
